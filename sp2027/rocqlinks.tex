% Links from the paper to the Rocq development in the anonymous artefact.
%
%   \rocq{name}          the Rocq name in \texttt, linked to its definition
%   \rocq[text]{name}    arbitrary text (e.g. a defined term), linked likewise
%   \rocqfile{path}      a repo-relative file or directory, linked
%   \rocqfile[shown]{path}  same, displaying `shown' (in \texttt) instead
%
% Locations live in the table below, one \rocqdef{name}{path}{line} per name.
% After updating the anonymous snapshot, refresh the line numbers with
%   python3 rocqlinks.py /path/to/SigmaProtocol
% A name missing from the table prints unlinked with a LaTeX warning.

\definecolor{rocqlink}{RGB}{0,72,150}

\ExplSyntaxOn
\tl_new:N \l__rocq_name_tl
\cs_generate_variant:Nn \tl_replace_all:Nnn { NV }
% print a name in \texttt, with hyphenat's breakable \_ for each underscore
\cs_new_protected:Npn \rocq@tt #1
  {
    \tl_set:Nx \l__rocq_name_tl { \tl_to_str:n {#1} }
    \tl_replace_all:NVn \l__rocq_name_tl \c_underscore_str { \_ }
    \texttt { \tl_use:N \l__rocq_name_tl }
  }
\ExplSyntaxOff
\newcommand{\rocqdef}[3]{%
  \if\relax\detokenize{#3}\relax
    \@namedef{rocq@url@#1}{#2}%
  \else
    \@namedef{rocq@url@#1}{#2\#L#3}%
  \fi}
\DeclareRobustCommand{\rocq}[2][]{%
  \@ifundefined{rocq@url@#2}%
    {\PackageWarning{rocqlinks}{No Rocq location for `\detokenize{#2}'}%
     \if\relax\detokenize{#1}\relax\rocq@tt{#2}\else#1\fi}%
    {\href{\rocqrepo/\@nameuse{rocq@url@#2}}%
      {\textcolor{rocqlink}{\if\relax\detokenize{#1}\relax\rocq@tt{#2}\else#1\fi}}}}
\DeclareRobustCommand{\rocqfile}[2][]{%
  \href{\rocqrepo/#2}%
    {\textcolor{rocqlink}{\if\relax\detokenize{#1}\relax\rocq@tt{#2}\else\rocq@tt{#1}\fi}}}
\makeatother
